[dec]
Nota1(float)
Nota2(float)
Media(float)
Situacao(integer)
[prog]
# Calcula a media de duas notas e classifica o aluno
read Nota1
read Nota2
Media << (Nota1 + Nota2) / 2

if Media >= 7.0 or Media == 10.0 then
start
	print "Aprovado"
	Situacao << 1
end
else
start
	if Media < 7.0 and Media >= 5.0 then
	start
		print "Recuperacao"
		Situacao << 2
	end
	else
	start
		print "Reprovado"
		Situacao << 0
	end
end

while Situacao <> 3 then
start
	print Situacao
	Situacao << Situacao + 1
end

print Media
